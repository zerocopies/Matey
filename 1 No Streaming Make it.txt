  provider,
  messages,
  tools = null,
  signal,
  onToken,
  onToolCallDelta,
  onFinish
}) {
  const body = {
    model: provider.model,
    messages,
    stream: true
  };
  if (tools) {
    body.tools = tools;
    body.tool_choice = 'auto';
  }

  const response = await fetch(`${provider.url}/v1/chat/completions`, {
    method: 'POST',

      'Content-Type': 'application/json',

    },
    body: JSON.stringify(body),
    signal
  });

  if (!response.ok) {
    const err = await response.text();
    throw new Error(`LLM error ${response.status}: ${err}`);
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';
  let fullContent = '';
  const toolCalls = new Map(); // index → tool call

  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';

      for (const line of lines) {
        const trimmed = line.trim();

        const data = trimmed.slice(6);
        if (data === '[DONE]') {
          onFinish?.({
            content: fullContent,
            tool_calls: Array.from(toolCalls.values())
          });
          return;
        }

        try {
          const json = JSON.parse(data);
          const delta = json.choices?.[0]?.delta;
          if (!delta) continue;

          if (delta.content) {
            fullContent += delta.content;
            onToken?.(delta.content);
          }

          if (delta.tool_calls) {
            for (const tc of delta.tool_calls) {

              if (!toolCalls.has(idx)) {
                toolCalls.set(idx, {
                  id: tc.id || `call_${idx}`,
                  type: 'function',
                  function: { name: '', arguments: '' }
                });

              if (tc.id) current.id = tc.id;
              if (tc.function?.name) current.function.name += tc.function.name;
              if (tc.function?.arguments) current.function.arguments += tc.function.arguments;
              onToolCallDelta?.(current);
            }
          }
        } catch (e) {

        }

JavaScriptfunction applySearchReplace(content, search, replace) {
  // Fast exact path
  if (content.includes(search)) {
    return content.replace(search, replace);
  }

  // Normalize whitespace version
  const normalize = s => s.replace(/\r\n/g, '\n').replace(/[ \t]+/g, ' ').trim();
  const normSearch = normalize(search);
  const lines = content.split('\n');

  // Simple sliding window fuzzy match (good enough + fast)
  for (let i = 0; i < lines.length; i++) {
    const window = lines.slice(i, i + search.split('\n').length).join('\n');
    if (normalize(window) === normSearch || similarity(normalize(window), normSearch) > 0.82) {
      const before = lines.slice(0, i).join('\n');
      const after = lines.slice(i + search.split('\n').length).join('\n');
      return (before ? before + '\n' : '') + replace + (after ? '\n' + after : '');
    }
  }

  // Last resort: return original + warning
  console.warn('SEARCH block not found even with fuzzy match');
  return content;
}

function similarity(a, b) {
  if (a === b) return 1;
  const longer = a.length > b.length ? a : b;
  const shorter = a.length > b.length ? b : a;
  if (longer.length === 0) return 1;
  return (longer.length - editDistance(longer, shorter)) / longer.length;
}

function editDistance(s1, s2) {
  // simple Levenshtein (you can optimize later)
  const costs = [];
  for (let i = 0; i <= s1.length; i++) {
    let lastValue = i;
    for (let j = 0; j <= s2.length; j++) {
      if (i === 0) costs[j] = j;
      else if (j > 0) {
        let newValue = costs[j - 1];
        if (s1.charAt(i - 1) !== s2.charAt(j - 1))
          newValue = Math.min(Math.min(newValue, lastValue), costs[j]) + 1;
        costs[j - 1] = lastValue;
        lastValue = newValue;
      }
    }
    if (i > 0) costs[s2.length] = lastValue;
  }
  return costs[s2.length];
}

    model: provider.model,
    messages,
    stream: true
  };

  if (tools && tools.length > 0) {
    body.tools = tools;
    body.tool_choice = 'auto';
  }

  let response;
  try {
    response = await fetch(`${provider.url.replace(/\/$/, '')}/v1/chat/completions`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${provider.key}`
      },
      body: JSON.stringify(body),
      signal
    });
  } catch (err) {
    if (err.name === 'AbortError') throw err;
    onError?.(err);
    throw err;
  }

  if (!response.ok) {
    const text = await response.text();
    const error = new Error(`LLM ${response.status}: ${text}`);
    onError?.(error);
    throw error;
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';
  let fullContent = '';
  const toolCallsMap = new Map();

  try {
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true });
      const lines = buffer.split('\n');
      buffer = lines.pop() || '';

      for (const line of lines) {
        const trimmed = line.trim();
        if (!trimmed.startsWith('data: ')) continue;

        const data = trimmed.slice(6);
        if (data === '[DONE]') {
          const finalToolCalls = Array.from(toolCallsMap.values());
          onFinish?.({
            content: fullContent,
            tool_calls: finalToolCalls.length > 0 ? finalToolCalls : undefined
          });
          return;
        }

        try {
          const json = JSON.parse(data);
          const delta = json.choices?.[0]?.delta;
          if (!delta) continue;

          if (delta.content) {
            fullContent += delta.content;
            onToken?.(delta.content);
          }

          if (delta.tool_calls) {
            for (const tcDelta of delta.tool_calls) {
              const idx = tcDelta.index ?? 0;
              if (!toolCallsMap.has(idx)) {
                toolCallsMap.set(idx, {
                  id: tcDelta.id || `call_${Date.now()}_${idx}`,
                  type: 'function',
                  function: { name: '', arguments: '' }
                });
              }
              const current = toolCallsMap.get(idx);
              if (tcDelta.id) current.id = tcDelta.id;

              if (tcDelta.function?.arguments) {

            }

        }

const STORE = 'sessions';

class HistoryStore {
  constructor() {
    this.dbPromise = this._open();
  }

  _open() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(DB_NAME, 1);
      request.onupgradeneeded = () => {
        const db = request.result;
        if (!db.objectStoreNames.contains(STORE)) {
          db.createObjectStore(STORE, { keyPath: 'workspaceId' });
        }
      };
      request.onsuccess = () => resolve(request.result);
      request.onerror = () => reject(request.error);
    });
  }

  async save(workspaceId, history) {
    if (!workspaceId) return;
    const db = await this.dbPromise;
    const tx = db.transaction(STORE, 'readwrite');
    const trimmed = Array.isArray(history) ? history.slice(-50) : [];
    tx.objectStore(STORE).put({
      workspaceId,
      history: trimmed,
      updatedAt: Date.now()
    });
    return new Promise((resolve, reject) => {
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error);
    });
  }

  async load(workspaceId) {
    if (!workspaceId) return [];
    const db = await this.dbPromise;
    return new Promise((resolve) => {
      const request = db.transaction(STORE).objectStore(STORE).get(workspaceId);
      request.onsuccess = () => {
        resolve(request.result?.history || []);
      };
      request.onerror = () => resolve([]);
    });
  }

  async clear(workspaceId) {
    const db = await this.dbPromise;
    const tx = db.transaction(STORE, 'readwrite');
    tx.objectStore(STORE).delete(workspaceId);
  }
}

JavaScript/**

export function applySearchReplace(content, searchBlock, replaceBlock) {
  if (!searchBlock) return content;

  // 1. Exact match (fast path)
  if (content.includes(searchBlock)) {
    return content.replace(searchBlock, replaceBlock);
  }

  // 2. Normalized whitespace match
  const norm = (str) => str.replace(/\r\n/g, '\n').replace(/[ \t]+/g, ' ').trim();
  const normSearch = norm(searchBlock);

  const lines = content.split('\n');
  const searchLines = searchBlock.split('\n');
  const windowSize = searchLines.length;

  for (let i = 0; i <= lines.length - windowSize; i++) {
    const window = lines.slice(i, i + windowSize).join('\n');
    if (norm(window) === normSearch) {
      const before = lines.slice(0, i).join('\n');
      const after = lines.slice(i + windowSize).join('\n');
      return [before, replaceBlock, after].filter(Boolean).join('\n');
    }
  }

  // 3. Light fuzzy fallback (similarity > 0.84)
  for (let i = 0; i <= lines.length - windowSize; i++) {
    const window = lines.slice(i, i + windowSize).join('\n');
    if (similarity(norm(window), normSearch) > 0.84) {
      const before = lines.slice(0, i).join('\n');
      const after = lines.slice(i + windowSize).join('\n');
      console.warn('[Matey] Applied fuzzy SEARCH/REPLACE');
      return [before, replaceBlock, after].filter(Boolean).join('\n');
    }
  }

  console.warn('[Matey] SEARCH block not found');
  return content;
}

function similarity(a, b) {
  if (a === b) return 1;
  const longer = a.length >= b.length ? a : b;
  const shorter = a.length >= b.length ? b : a;
  if (longer.length === 0) return 1.0;
  return (longer.length - levenshtein(longer, shorter)) / longer.length;
}

function levenshtein(s1, s2) {
  const prev = new Array(s2.length + 1);
  const curr = new Array(s2.length + 1);
  for (let j = 0; j <= s2.length; j++) prev[j] = j;

  for (let i = 1; i <= s1.length; i++) {
    curr[0] = i;
    for (let j = 1; j <= s2.length; j++) {
      const cost = s1[i - 1] === s2[j - 1] ? 0 : 1;
      curr[j] = Math.min(curr[j - 1] + 1, prev[j] + 1, prev[j - 1] + cost);
    }
    for (let j = 0; j <= s2.length; j++) prev[j] = curr[j];
  }
  return prev[s2.length];
}
